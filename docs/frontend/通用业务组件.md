## 三、JsonViewer - JSON 树形展示

### 3.1 实现

```vue
<!-- src/components/common/JsonViewer.vue -->
<template>
  <div class="json-viewer" :class="{ 'is-dark': isDark }">
    <div class="json-toolbar">
      <el-button text size="small" @click="expandAll">
        <i class="fas fa-expand" /> 全部展开
      </el-button>
      <el-button text size="small" @click="collapseAll">
        <i class="fas fa-compress" /> 全部收起
      </el-button>
      <el-button text size="small" @click="copy">
        <i class="fas fa-copy" /> 复制
      </el-button>
    </div>
    <el-scrollbar>
      <pre class="json-content"><code v-html="renderedHtml" /></pre>
    </el-scrollbar>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue';
import { ElMessage } from 'element-plus';

interface Props {
  value: any;
  isDark?: boolean;
  initiallyExpanded?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  isDark: false,
  initiallyExpanded: true,
});

const collapsedPaths = ref(new Set<string>());

const renderedHtml = computed(() => buildHtml(props.value, '', props.initiallyExpanded));

function buildHtml(value: any, path: string, expanded: boolean): string {
  if (value === null) return '<span class="json-null">null</span>';
  if (typeof value === 'boolean') return `<span class="json-bool">${value}</span>`;
  if (typeof value === 'number') return `<span class="json-number">${value}</span>`;
  if (typeof value === 'string') return `<span class="json-string">"${escapeHtml(value)}"</span>`;

  const isArray = Array.isArray(value);
  const items = isArray ? value : Object.entries(value);
  const isEmpty = isArray ? !value.length : !Object.keys(value).length;
  if (isEmpty) return isArray ? '[]' : '{}';

  const isCollapsed = collapsedPaths.value.has(path);
  const open = isArray ? '[' : '{';
  const close = isArray ? ']' : '}';
  const toggle = `<span class="json-toggle" data-path="${path}">${isCollapsed ? '▶' : '▼'}</span>`;

  if (isCollapsed) {
    const summary = isArray
      ? `Array(${(value as any[]).length})`
      : `Object{${Object.keys(value).length}}`;
    return `${toggle}${open}<span class="json-summary"> ${summary} </span>${close}`;
  }

  const children = isArray
    ? (value as any[]).map((v, i) => buildHtml(v, `${path}[${i}]`, expanded))
    : (items as [string, any][]).map(([k, v]) =>
        `<span class="json-key">"${k}"</span>: ${buildHtml(v, `${path}.${k}`, expanded)}`
      );

  return `${toggle}${open}\n  ${children.join(',\n  ')}\n${close}`;
}

function escapeHtml(s: string) {
  return s.replace(/[&<>"']/g, c => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]!));
}

function expandAll() { collapsedPaths.value.clear(); }
function collapseAll() {
  // 收集所有路径
  const all = new Set<string>();
  function walk(v: any, p: string) {
    if (v && typeof v === 'object') {
      all.add(p);
      const items = Array.isArray(v) ? v.entries() : Object.entries(v);
      if (Array.isArray(v)) v.forEach((c, i) => walk(c, `${p}[${i}]`));
      else Object.entries(v).forEach(([k, c]) => walk(c, `${p}.${k}`));
    }
  }
  walk(props.value, '');
  collapsedPaths.value = all;
}

async function copy() {
  await navigator.clipboard.writeText(JSON.stringify(props.value, null, 2));
  ElMessage.success('已复制');
}
</script>

<style scoped lang="scss">
.json-viewer {
  border: 1px solid var(--border);
  border-radius: 6px;
  background: var(--bg-primary);
  &.is-dark { background: #1e293b; color: #f1f5f9; }
}
.json-toolbar {
  display: flex; gap: 4px; padding: 6px 8px;
  border-bottom: 1px solid var(--border);
}
.json-content {

```

## 四、CronEditor - Cron 表达式编辑器

### 4.1 封装

```vue
<!-- src/components/common/CronEditor.vue -->
<template>
  <div class="cron-editor">
    <el-tabs v-model="mode">
      <el-tab-pane label="可视化" name="visual">
        <vue-cron-editor
          v-model="cronValue"
          :disabled="readonly"
          @change="onChange"
        />
      </el-tab-pane>
      <el-tab-pane label="表达式" name="expr">
        <el-input
          v-model="cronValue"
          placeholder="0 0 * * * ?"
          font-family="monospace"
          @input="onChange"
        >
          <template #append>
            <el-button @click="validate">校验</el-button>
          </template>
        </el-input>
      </el-tab-pane>
    </el-tabs>
    <div class="cron-preview" v-if="cronValue">
      <i class="fas fa-calendar-alt" />
      下次执行：
      <strong>{{ nextRunTimes.join('、') }}</strong>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, watch } from 'vue';
import VueCronEditor from 'vue-cron-editor';
import cronParser from 'cron-parser';
import { ElMessage } from 'element-plus';

interface Props {
  modelValue: string;
  readonly?: boolean;
}

const props = defineProps<Props>();
const emit = defineEmits<{ (e: 'update:modelValue', v: string): void }>();

const cronValue = ref(props.modelValue);
const mode = ref<'visual' | 'expr'>('visual');

const nextRunTimes = ref<string[]>([]);

watch(() => props.modelValue, (v) => { cronValue.value = v; });

function onChange(v: string) {
  emit('update:modelValue', v);
  computeNext(v);
}

function validate() {
  try {
    cronParser.parseExpression(cronValue.value);
    ElMessage.success('Cron 表达式有效');
    computeNext(cronValue.value);
  } catch (e: any) {
    ElMessage.error('无效的 Cron 表达式');
  }
}

function computeNext(expr: string) {
  try {
    const interval = cronParser.parseExpression(expr, { currentDate: new Date() });
    nextRunTimes.value = [];
    for (let i = 0; i < 5; i++) {
      nextRunTimes.value.push(interval.next().toDate().toLocaleString('zh-CN'));
    }
  } catch {
    nextRunTimes.value = [];
  }
}

computeNext(props.modelValue);
</script>

<style scoped lang="scss">
.cron-editor {
  :deep(.vue-cron-editor) { border: 1px solid var(--border); border-radius: 6px; }
}
.cron-preview {
  margin-top: 12px; padding: 8px 12px;
  background: var(--info-bg); border-radius: 6px;
  font-size: 12px; color: var(--text-secondary);
}
</style>
```

> 注：`cron-parser` 需安装：`pnpm add cron-parser @types/cron-parser`

```typescript
// src/plugins/markdown.ts
import MarkdownIt from 'markdown-it';
import hljs from 'highlight.js';
import 'highlight.js/styles/github.css';

const md = new MarkdownIt({
  html: false,
  linkify: true,
  typographer: true,
  highlight: (str, lang) => {
    if (lang && hljs.getLanguage(lang)) {
      try {
        return `<pre class="hljs"><code>${
          hljs.highlight(str, { language: lang, ignoreIllegals: true }).value
        }</code></pre>`;
      } catch {}
    }
    return `<pre class="hljs"><code>${md.utils.escapeHtml(str)}</code></pre>`;
  },
});

export function renderMarkdown(text: string): string {
  return md.render(text);
}
```

```typescript
// src/composables/useConfirm.ts
import { ElMessageBox, ElMessageBoxOptions } from 'element-plus';

export function useConfirm() {
  function confirm(options: {
    title?: string;
    message: string;
    type?: 'warning' | 'info' | 'success' | 'error';
    confirmText?: string;
    cancelText?: string;
    danger?: boolean;
  }): Promise<void> {
    const opts: ElMessageBoxOptions = {
      title: options.title || '确认',
      message: options.message,
      type: options.type || 'warning',
      confirmButtonText: options.confirmText || '确定',
      cancelButtonText: options.cancelText || '取消',
      confirmButtonClass: options.danger ? 'el-button--danger' : '',
    };
    return ElMessageBox.confirm(options.message, options.title || '确认', opts) as any;
  }

  return { confirm };
}
```

## 十、ThemeSwitcher - 主题切换

```typescript
// src/composables/useTheme.ts
import { ref, watch } from 'vue';

export type Theme = 'light' | 'dark';

const theme = ref<Theme>((localStorage.getItem('theme') as Theme) || 'light');

watch(theme, (t) => {
  document.documentElement.classList.toggle('dark', t === 'dark');
  localStorage.setItem('theme', t);
}, { immediate: true });

export function useTheme() {
  function toggle() {
    theme.value = theme.value === 'light' ? 'dark' : 'light';
  }
  return { theme, toggle };
}
```

```vue
<script setup lang="ts">
import { ref, watch } from 'vue';
import { ElMessage } from 'element-plus';

const visible = defineModel<boolean>({ default: false });
const emit = defineEmits<{ (e: 'import', data: any): void }>();

const mode = ref<'file' | 'paste' | 'url'>('file');
const file = ref<File | null>(null);
const pastedText = ref('');
const url = ref('');
const parsed = ref<any>(null);

function onFileChange(f: any) {
  file.value = f.raw;
  readFile(f.raw);
}

function readFile(f: File) {
  const reader = new FileReader();
  reader.onload = (e) => {
    pastedText.value = e.target?.result as string;
    tryParse(pastedText.value);
  };
  reader.readAsText(f);
}

watch(pastedText, (v) => tryParse(v));

function tryParse(text: string) {
  if (!text.trim()) { parsed.value = null; return; }
  try {
    parsed.value = JSON.parse(text);
  } catch (e: any) {
    parsed.value = null;
    ElMessage.warning('JSON 格式错误');
  }
}

async function fetchFromUrl() {
  try {
    const res = await fetch(url.value);
    const text = await res.text();
    pastedText.value = text;
    mode.value = 'paste';
  } catch (e: any) {
    ElMessage.error('获取失败：' + e.message);
  }
}

function onConfirm() {
  emit('import', parsed.value);
  visible.value = false;
  ElMessage.success('导入成功');
}
```

```vue
<!-- src/components/common/ExportDialog.vue -->
<template>
  <el-dialog v-model="visible" title="导出工作流" width="480px">
    <el-form label-position="top">
      <el-form-item label="导出格式">
        <el-radio-group v-model="format">
          <el-radio-button label="json">JSON</el-radio-button>
          <el-radio-button label="yaml">YAML</el-radio-button>
          <el-radio-button label="png">PNG 图片</el-radio-button>
          <el-radio-button label="zip">ZIP（含依赖）</el-radio-button>
        </el-radio-group>
      </el-form-item>

      <el-form-item v-if="format === 'json' || format === 'yaml'">
        <el-checkbox v-model="includeCredentials">包含凭据信息（API Key 等）</el-checkbox>
        <el-checkbox v-model="minify">压缩为单行</el-checkbox>
      </el-form-item>
    </el-form>

    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" @click="onExport">
        <i class="fas fa-download" /> 导出
      </el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { saveAs } from 'file-saver';
import yaml from 'js-yaml';
import html2canvas from 'html2canvas';

const visible = defineModel<boolean>({ default: false });
const props = defineProps<{ workflow: any; canvasEl?: HTMLElement }>();

const format = ref<'json' | 'yaml' | 'png' | 'zip'>('json');
const includeCredentials = ref(false);
const minify = ref(false);

async function onExport() {
  const data = { ...props.workflow };
  if (!includeCredentials.value) stripCredentials(data);

  switch (format.value) {
    case 'json': {
      const text = minify.value ? JSON.stringify(data) : JSON.stringify(data, null, 2);
      saveAs(new Blob([text], { type: 'application/json' }), `${data.name}.json`);
      break;
    }
    case 'yaml': {
      const text = yaml.dump(data);
      saveAs(new Blob([text], { type: 'text/yaml' }), `${data.name}.yaml`);
      break;
    }
    case 'png': {
      if (!props.canvasEl) return;
      const canvas = await html2canvas(props.canvasEl);
      canvas.toBlob(blob => blob && saveAs(blob, `${data.name}.png`));
      break;
    }
    case 'zip': {
      // 使用 jszip 打包
      break;
    }
  }
  visible.value = false;
}

function stripCredentials(obj: any) {
  // 递归去除敏感字段
  if (obj.httpConfig?.headers?.Authorization) delete obj.httpConfig.headers.Authorization;
  if (obj.apiKey) obj.apiKey = '***';
}
</script>
```
