### 2.3 统计图表组件

```vue
<!-- src/components/stat/ExecTrendChart.vue -->
<template>
  <div class="exec-trend-chart" ref="chartRef" />
</template>

<script setup lang="ts">
import { ref, onMounted, watch } from 'vue';
import * as echarts from 'echarts';

interface Props {
  data: {
    date: string;
    executions: number;
    successRate: number;
  }[];
}

const props = defineProps<Props>();
const chartRef = ref<HTMLElement>();

onMounted(() => {
  initChart();
});

watch(() => props.data, () => {
  initChart();
});

function initChart() {
  if (!chartRef.value) return;

  const chart = echarts.init(chartRef.value);

  const option = {
    tooltip: {
      trigger: 'axis',
    },
    legend: {
      data: ['执行次数', '成功率'],
    },
    grid: {
      left: '3%',
      right: '4%',
      bottom: '3%',
      containLabel: true,
    },
    xAxis: {
      type: 'category',
      data: props.data.map(d => d.date),
    },
    yAxis: [
      {
        type: 'value',
        name: '执行次数',
      },
      {
        type: 'value',
        name: '成功率',
        min: 0,
        max: 100,
        axisLabel: {
          formatter: '{value}%',
        },
      },
    ],
    series: [
      {
        name: '执行次数',
        type: 'bar',
        data: props.data.map(d => d.executions),
        itemStyle: { color: '#3b82f6' },
      },
      {
        name: '成功率',
        type: 'line',
        yAxisIndex: 1,
        data: props.data.map(d => d.successRate),
        smooth: true,
        itemStyle: { color: '#10b981' },
      },
    ],
  };

  chart.setOption(option);
}
</script>

<style scoped>
.exec-trend-chart {
  width: 100%;
  height: 300px;
}
</style>
```

```vue
<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue';
import { teamApi } from '@/api/team';
import { useUserStore } from '@/stores/user';
import { ElMessage, ElMessageBox } from 'element-plus';

const userStore = useUserStore();

const currentUserId = ref('');

// 权限设置
const permissions = reactive({
  workflow: 'all',
  template: 'all',
});

const inviteRole = ref('member');

  team.value = await teamApi.getCurrentTeam();
  members.value = await teamApi.getMembers(team.value.id);

async function handleInvite() {
  await teamApi.inviteMember({
    teamId: team.value.id,
    email: inviteEmail.value,
    role: inviteRole.value,

  ElMessage.success('邀请已发送');
  showInviteDialog.value = false;
  members.value = await teamApi.getMembers(team.value.id);
}

async function handleEditRole(member: any) {
  try {
    await ElMessageBox.confirm(
      `确定要将 ${member.username} 的角色修改为管理员？`,
      '修改角色',
      { type: 'warning' }
    );

    await teamApi.updateMemberRole(member.id, 'admin');
    ElMessage.success('修改成功');
    members.value = await teamApi.getMembers(team.value.id);
  } catch {
    // 用户取消
  }
}

async function handleRemove(member: any) {
  try {
    await ElMessageBox.confirm(
      `确定要将 ${member.username} 移出团队？`,
      '移除成员',
      { type: 'warning' }
    );

    await teamApi.removeMember(member.id);
    ElMessage.success('已移出');
    members.value = await teamApi.getMembers(team.value.id);
  } catch {
    // 用户取消
  }
}

function formatDate(date: string): string {
  return new Date(date).toLocaleDateString('zh-CN');
}

function formatRelativeTime(date: string): string {
  const diff = Date.now() - new Date(date).getTime();
  const minutes = Math.floor(diff / 60000);
  if (minutes < 1) return '刚刚';
  if (minutes < 60) return `${minutes}分钟前`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}小时前`;
  const days = Math.floor(hours / 24);
  return `${days}天前`;
}
</script>
```
