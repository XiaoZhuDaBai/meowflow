## 一、流程概述

### 1.1 用户操作路径

```
执行工作流（Webhook/测试运行） → 查看执行日志 → 查看执行详情
```

### 1.2 涉及的核心组件

| 组件 | 文件路径 | 职责 |
|------|----------|------|
| `LogPanel.vue` | `src/components/workflow/LogPanel.vue` | 日志面板组件 |
| `List.vue` | `src/views/log/List.vue` | 执行日志列表页面 |
| `ExecutionDetail.vue` | `src/views/log/Detail.vue` | 执行详情页面 |
| `JsonViewer.vue` | `src/components/common/JsonViewer.vue` | JSON 查看器 |

### 1.3 涉及的数据模型

| 模型 | 文件路径 | 说明 |
|------|----------|------|
| `Execution` | `src/types/execution.ts` | 执行记录实体 |
| `NodeExecution` | `src/types/execution.ts` | 节点执行记录 |
| `ExecutionLog` | `src/types/execution.ts` | 执行日志 |

#### 2.2.1 测试运行流程

```typescript
// src/views/workflow/Editor.vue
async function handleTest() {
  // 1. 打开日志面板
  showLogPanel.value = true;

  // 2. 清空日志
  executionLogs.value = [];

  // 3. 获取测试输入
  const testInput = await promptTestInput();

  // 4. 调用执行 API
  try {
    addLog({
      level: 'info',
      message: '开始测试运行...',
      timestamp: Date.now(),
    });

    const result = await workflowApi.execute(workflowId.value, {
      input: testInput,
    });

    // 5. 更新节点状态
    updateNodeStatuses(result.nodes);

    // 6. 添加完成日志
    addLog({
      level: result.success ? 'success' : 'error',
      message: result.success ? '执行成功' : '执行失败',
      timestamp: Date.now(),
    });
  } catch (error: any) {
    addLog({
      level: 'error',
      message: `执行失败: ${error.message}`,
      timestamp: Date.now(),
    });
  }
}

// 更新节点执行状态
function updateNodeStatuses(nodes: NodeExecution[]) {
  nodes.forEach(node => {
    const nodeStatus = node.status;
    workflowStore.updateNode(node.id, { status: nodeStatus });

    addLog({
      level: nodeStatus === 'success' ? 'success' : 'error',
      nodeName: node.nodeName,
      message: nodeStatus === 'success'
        ? `执行成功 (${node.duration}ms)`
        : `执行失败: ${node.error}`,
      timestamp: Date.now(),
    });
  });
}
```

## 四、实时日志（SSE）

### 4.1 SSE 服务封装

```typescript
// src/api/sse.ts
export class ExecutionSSE {
  private eventSource: EventSource | null = null;
  private onMessage: (log: ExecutionLog) => void;
  private onComplete: (result: ExecutionResult) => void;
  private onError: (error: Error) => void;

  constructor(options: {
    executionId: string;
    onMessage: (log: ExecutionLog) => void;
    onComplete: (result: ExecutionResult) => void;
    onError: (error: Error) => void;
  }) {
    this.onMessage = options.onMessage;
    this.onComplete = options.onComplete;
    this.onError = options.onError;

    const token = localStorage.getItem('access_token');
    this.eventSource = new EventSource(
      `${import.meta.env.VITE_API_BASE_URL}/executions/${options.executionId}/stream?token=${token}`
    );

    this.eventSource.onmessage = this.handleMessage.bind(this);
    this.eventSource.onerror = this.handleError.bind(this);
  }

  private handleMessage(event: MessageEvent) {
    const data = JSON.parse(event.data);

    if (data.type === 'log') {
      this.onMessage(data.log);
    } else if (data.type === 'complete') {
      this.onComplete(data.result);
      this.close();
    }
  }

  private handleError(event: Event) {
    this.onError(new Error('SSE 连接错误'));
    this.close();
  }

  close() {
    this.eventSource?.close();
    this.eventSource = null;
  }
}
```

```typescript
import { defineStore } from 'pinia';

  async function fetchExecutionDetail(id: string): Promise<Execution> {
    loading.value = true;
    try {
      currentExecution.value = await executionApi.getById(id);
      return currentExecution.value;
    } finally {
      loading.value = false;
    }
  }

  async function retryExecution(id: string): Promise<Execution> {
    return await executionApi.retry(id);
  }

  return {
    executions,
    currentExecution,
    loading,
    total,
    fetchExecutions,
    fetchExecutionDetail,
    retryExecution,
  };
});
```
