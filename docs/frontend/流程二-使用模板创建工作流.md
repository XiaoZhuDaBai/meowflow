# 流程二：使用模板创建工作流

> 详细的前端实现文档，描述使用模板市场快速创建工作流的完整流程
> 版本：v1.0 | 更新日期：2026-07-17

## 一、流程概述

### 1.1 用户操作路径

```
进入模板市场 → 浏览/搜索模板 → 预览模板详情 → 使用模板创建 → 跳转编辑器修改 → 保存发布
```

### 1.2 涉及的核心组件

| 组件 | 文件路径 | 职责 |
|------|----------|------|
| `Market.vue` | `src/views/template/Market.vue` | 模板市场主页面 |
| `TemplateCard.vue` | `src/components/template/TemplateCard.vue` | 模板卡片组件 |
| `TemplatePreview.vue` | `src/components/template/TemplatePreview.vue` | 模板详情预览弹窗 |
| `TemplateFilter.vue` | `src/components/template/TemplateFilter.vue` | 模板筛选组件 |

### 1.3 涉及的数据模型

| 模型 | 文件路径 | 说明 |
|------|----------|------|
| `Template` | `src/api/template.ts` | 模板实体 |
| `Workflow` | `src/types/workflow.ts` | 工作流实体 |

```vue
<!-- src/components/template/TemplateFilter.vue -->
<template>
  <div class="template-filter">
    <!-- 搜索框 -->
    <el-input
      :model-value="keyword"
      placeholder="搜索模板..."
      clearable

    >
      <el-option value="popular" label="热门优先" />
      <el-option value="recent" label="最新优先" />
      <el-option value="rating" label="评分优先" />
    </el-select>

    <!-- 标签筛选 -->
    <el-select
      v-model="localTags"
      multiple
      placeholder="选择标签"
      collapse-tags
      @change="handleTagsChange"
    >
      <el-option
        v-for="tag in availableTags"
        :key="tag"
        :value="tag"
        :label="tag"
      />
    </el-select>
  </div>
</template>

<script setup lang="ts">
import { ref, watch } from 'vue';

interface Props {
  keyword?: string;
  sort?: 'popular' | 'recent' | 'rating';
  tags?: string[];
  availableTags: string[];
}

const props = withDefaults(defineProps<Props>(), {
  keyword: '',
  sort: 'popular',
  tags: () => [],
});

const emit = defineEmits<{
  (e: 'update:keyword', value: string): void;
  (e: 'update:sort', value: string): void;
  (e: 'update:tags', value: string[]): void;
  (e: 'search'): void;
}>();

const localTags = ref(props.tags);

function handleKeywordChange(value: string) {
  emit('update:keyword', value);
  emit('search');
}

function handleSortChange(value: string) {
  emit('update:sort', value as any);
  emit('search');
}

function handleTagsChange(value: string[]) {
  emit('update:tags', value);
  emit('search');
}
</script>
```

### 步骤 4 - 使用模板创建

#### 2.4.1 使用模板流程

```typescript
// src/views/template/Market.vue
async function handleUseTemplate(template: Template) {
  try {
    // 1. 显示加载状态
    loading.value = true;

    // 2. 调用 API 使用模板创建工作流
    const workflow = await templateStore.useTemplate(template.id);

    // 3. 关闭预览弹窗
    previewVisible.value = false;

    // 4. 跳转到编辑器
    router.push(`/editor/${workflow.id}`);

    // 5. 提示用户
    ElMessage.success('模板创建成功，开始编辑吧！');
  } catch (error: any) {
    ElMessage.error(`创建失败: ${error.message}`);
  } finally {
    loading.value = false;
  }
}
```

#### 2.4.2 模板 Store

```typescript
// src/stores/template.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import { templateApi, type Template } from '@/api/template';

export const useTemplateStore = defineStore('template', () => {
  // State
  const templates = ref<Template[]>([]);
  const currentTemplate = ref<Template | null>(null);
  const categories = ref<{ value: string; label: string; count: number }[]>([]);
  const total = ref(0);
  const loading = ref(false);

  // Getters
  const officialTemplates = computed(() =>
    templates.value.filter(t => t.isOfficial)
  );

  const popularTemplates = computed(() =>
    [...templates.value].sort((a, b) => b.usageCount - a.usageCount)
  );

  // Actions
  async function fetchTemplates(params?: {
    category?: string;
    keyword?: string;
    sort?: 'popular' | 'recent' | 'rating';
    tags?: string[];
    page?: number;
    pageSize?: number;
  }) {
    loading.value = true;
    try {
      const result = await templateApi.getList(params);
      templates.value = result.items;
      total.value = result.total;
    } finally {
      loading.value = false;
    }
  }

  async function fetchTemplate(id: string) {
    loading.value = true;
    try {
      currentTemplate.value = await templateApi.getById(id);
    } finally {
      loading.value = false;
    }
  }

  async function fetchCategories() {
    categories.value = await templateApi.getCategories();
  }

  async function useTemplate(templateId: string): Promise<any> {
    // 调用 API 使用模板创建工作流
    return await templateApi.createFromTemplate(templateId);
  }

  return {
    templates,
    currentTemplate,
    categories,
    total,
    loading,
    officialTemplates,
    popularTemplates,
    fetchTemplates,
    fetchTemplate,
    fetchCategories,
    useTemplate,
  };
});
```

## 三、模板 API

### 3.1 API 定义

```typescript
// src/api/template.ts
import request from './request';

export interface Template {
  id: string;
  name: string;
  description: string;
  category: string;
  tags: string[];
  coverImage?: string;
  author: string;
  usageCount: number;
  rating: number;
  workflow: {
    nodes: any[];
    edges: any[];
  };
  isOfficial: boolean;
  createdAt: string;
  updatedAt: string;
}

export const templateApi = {
  /**
   * 获取模板列表
   */
  getList(params?: {
    category?: string;
    keyword?: string;
    sort?: 'popular' | 'recent' | 'rating';
    tags?: string[];
    page?: number;
    pageSize?: number;
  }): Promise<{ items: Template[]; total: number }> {
    return request.get('/templates', { params });
  },

  /**
   * 获取模板详情
   */
  getById(id: string): Promise<Template> {
    return request.get(`/templates/${id}`);
  },

  /**
   * 使用模板创建工作流
   */
  createFromTemplate(id: string, name?: string): Promise<any> {
    return request.post(`/templates/${id}/use`, { name });
  },

  /**
   * 获取官方模板
   */
  getOfficial(): Promise<Template[]> {
    return request.get('/templates/official');
  },

  /**
   * 获取热门模板
   */
  getPopular(limit?: number): Promise<Template[]> {
    return request.get('/templates/popular', { params: { limit } });
  },

  /**
   * 获取分类列表
   */
  getCategories(): Promise<{ value: string; label: string; count: number }[]> {
    return request.get('/templates/categories');
  },
};
```
