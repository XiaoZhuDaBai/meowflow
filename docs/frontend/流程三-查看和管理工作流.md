# 流程三：查看和管理工作流

> 详细的前端实现文档，描述查看、编辑、复制、删除工作流的完整流程
> 版本：v1.0 | 更新日期：2026-07-17

## 一、流程概述

### 1.1 用户操作路径

```
进入工作流列表 → 查看工作流卡片 → 进入详情页/编辑器 → 执行管理操作
```

### 1.2 涉及的核心组件

| 组件 | 文件路径 | 职责 |
|------|----------|------|
| `List.vue` | `src/views/workflow/List.vue` | 工作流列表页面 |
| `Detail.vue` | `src/views/workflow/Detail.vue` | 工作流详情页面 |
| `WorkflowItem.vue` | `src/components/workflow/WorkflowItem.vue` | 工作流卡片组件 |
| `StatusTag.vue` | `src/components/common/StatusTag.vue` | 状态标签组件 |

### 1.3 涉及的数据模型

| 模型 | 文件路径 | 说明 |
|------|----------|------|
| `WorkflowListItem` | `src/types/workflow.ts` | 工作流列表项 |
| `Workflow` | `src/types/workflow.ts` | 完整工作流实体 |
| `WorkflowStats` | `src/types/workflow.ts` | 工作流统计信息 |

async function handlePublish(workflow: WorkflowListItem) {
  try {
    await workflowStore.publishWorkflow(workflow.id);

    // 更新列表中的状态
    const index = workflows.value.findIndex(w => w.id === workflow.id);
    if (index !== -1) {

## 三、Store 实现

### 3.1 工作流 Store

```typescript
// src/stores/workflow.ts
import { defineStore } from 'pinia';
import { ref } from 'vue';
import { workflowApi } from '@/api/workflow';

export const useWorkflowStore = defineStore('workflow', () => {
  // State
  const workflows = ref<WorkflowListItem[]>([]);
  const currentWorkflow = ref<Workflow | null>(null);
  const loading = ref(false);
  const total = ref(0);

  // Actions
  async function fetchWorkflows(params?: {
    page?: number;
    pageSize?: number;
    keyword?: string;
    status?: string;
    category?: string;
  }) {
    loading.value = true;
    try {
      const result = await workflowApi.getList(params);
      workflows.value = result.items;
      total.value = result.total;
      return result;
    } finally {
      loading.value = false;
    }
  }

  async function fetchWorkflowDetail(id: string): Promise<Workflow> {
    loading.value = true;
    try {
      currentWorkflow.value = await workflowApi.getById(id);
      return currentWorkflow.value;
    } finally {
      loading.value = false;
    }
  }

  async function deleteWorkflow(id: string): Promise<void> {
    await workflowApi.delete(id);
  }

  async function duplicateWorkflow(id: string): Promise<Workflow> {
    return await workflowApi.duplicate(id);
  }

  async function publishWorkflow(id: string): Promise<void> {
    await workflowApi.publish(id);
  }

  async function unpublishWorkflow(id: string): Promise<void> {
    await workflowApi.unpublish(id);
  }

  return {
    workflows,
    currentWorkflow,
    loading,
    total,
    fetchWorkflows,
    fetchWorkflowDetail,
    deleteWorkflow,
    duplicateWorkflow,
    publishWorkflow,
    unpublishWorkflow,
  };
});
```
