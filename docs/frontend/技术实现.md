
## 五、工作流画布实现

### 5.1 画布组件核心

```vue
<!-- src/components/workflow/canvas/WorkflowCanvas.vue -->
<template>
  <div ref="containerRef" class="workflow-canvas">
    <!-- G6 图例容器 -->
    <div id="g6-container" class="g6-container" />

    <!-- 工具栏 -->
    <CanvasToolbar
      v-if="!readonly"
      class="canvas-toolbar"
      @undo="undo"
      @redo="redo"
      @zoom-in="zoomIn"
      @zoom-out="zoomOut"
      @fit-view="fitView"
      @add-node="openNodePalette"
      @run="testRun"
      @publish="publish"
    />

    <!-- 节点选择面板 -->
    <NodePalette
      v-if="showPalette"
      ref="paletteRef"
      @select="addNode"
      @close="closeNodePalette"
    />

    <!-- 空状态 -->
    <div v-if="nodes.length === 0" class="empty-state">
      <div class="empty-icon">
        <i class="fas fa-project-diagram" />
      </div>
      <h3>开始创建你的第一个工作流</h3>
      <p>拖拽节点到画布，或使用模板快速开始</p>
      <el-button type="primary" @click="openNodePalette">
        <i class="fas fa-plus" />
        新建工作流
      </el-button>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onBeforeUnmount, watch } from 'vue';
import { Graph } from '@antv/g6';
import { useWorkflowStore } from '@/stores/workflow';
import { useCanvasStore } from '@/stores/canvas';
import CanvasToolbar from './CanvasToolbar.vue';
import NodePalette from '../NodePalette.vue';
import type { WorkflowNode, WorkflowEdge } from '@/types';

interface Props {
  readonly?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  readonly: false,
});

const emit = defineEmits<{
  (e: 'change', nodes: WorkflowNode[], edges: WorkflowEdge[]): void;
  (e: 'run'): void;
  (e: 'publish'): void;
}>();

// Stores
const workflowStore = useWorkflowStore();
const canvasStore = useCanvasStore();

// Refs
const containerRef = ref<HTMLElement>();
const paletteRef = ref<InstanceType<typeof NodePalette>>();
const showPalette = ref(false);

// G6 Graph 实例
let graph: Graph | null = null;

// 初始化 G6
function initGraph() {
  if (!containerRef.value) return;

  graph = new Graph({
    container: 'g6-container',
    width: containerRef.value.offsetWidth,
    height: containerRef.value.offsetHeight,
    modes: {
      default: ['drag-canvas', 'zoom-canvas', 'drag-node'],
    },
    defaultNode: {
      type: 'rect',
      style: {
        fill: '#ffffff',
        stroke: '#e2e8f0',
        radius: 8,
        shadowColor: 'rgba(0,0,0,0.1)',
        shadowBlur: 6,
        shadowOffsetY: 4,
      },
    },
    defaultEdge: {
      type: 'polyline',
      style: {
        stroke: '#e2e8f0',
        lineWidth: 2,
        endArrow: true,
      },
    },
    nodeStateStyles: {
      hover: {
        shadowColor: 'rgba(99,102,241,0.3)',
        shadowBlur: 12,
      },
      selected: {
        stroke: '#6366f1',
        lineWidth: 2,
        shadowColor: 'rgba(99,102,241,0.3)',
        shadowBlur: 12,
      },
    },
  });

  // 绑定事件
  graph.on('node:click', handleNodeClick);
  graph.on('node:dragend', handleNodeDragEnd);
  graph.on('blank:click', handleBlankClick);

  // 渲染初始数据
  renderGraph();
}

// 渲染图数据
function renderGraph() {
  if (!graph) return;

  const data = {
    nodes: nodes.value.map(transformNode),
    edges: edges.value.map(transformEdge),
  };

  graph.data(data);
  graph.render();
}

// 转换节点数据给 G6
function transformNode(node: WorkflowNode) {
  return {
    id: node.id,
    type: 'custom-node',
    data: node,
    style: {
      x: node.position.x,
      y: node.position.y,
    },
  };
}

// 转换边数据给 G6
function transformEdge(edge: WorkflowEdge) {
  return {
    id: edge.id,
    source: edge.source,
    target: edge.target,
    sourceHandle: edge.sourceHandle,
    targetHandle: edge.targetHandle,
    label: edge.label,
  };
}

// 节点点击
function handleNodeClick(e: any) {
  const nodeId = e.item.getID();
  canvasStore.selectNode(nodeId);
  emit('node-select', nodeId);
}

// 节点拖拽结束
function handleNodeDragEnd(e: any) {
  const nodeId = e.item.getID();
  const position = e.item.getModel();

  workflowStore.updateNodePosition(nodeId, {
    x: position.x,
    y: position.y,
  });
}

// 空白区域点击
function handleBlankClick() {
  canvasStore.clearSelection();
}

// 添加节点
function addNode(nodeConfig: Partial<WorkflowNode>) {
  const newNode: WorkflowNode = {
    id: `node-${Date.now()}`,
    type: nodeConfig.type!,
    category: nodeConfig.category!,
    title: nodeConfig.title!,
    position: nodeConfig.position || { x: 400, y: 200 },
    config: nodeConfig.config || {},
  };

  workflowStore.addNode(newNode);
  showPalette.value = false;
  renderGraph();
}

// 画布操作
function undo() { workflowStore.undo(); }
function redo() { workflowStore.redo(); }
function zoomIn() { graph?.zoom(1.1); }
function zoomOut() { graph?.zoom(0.9); }
function fitView() { graph?.fitView(); }
function openNodePalette() { showPalette.value = true; }
function closeNodePalette() { showPalette.value = false; }
function testRun() { emit('run'); }
function publish() { emit('publish'); }

// 生命周期
onMounted(() => {
  initGraph();
  window.addEventListener('resize', handleResize);
});

onBeforeUnmount(() => {
  graph?.destroy();
  window.removeEventListener('resize', handleResize);
});

// 响应式
watch(() => props.nodes, renderGraph, { deep: true });
</script>

<style scoped lang="scss">
.workflow-canvas {
  position: relative;
  width: 100%;
  height: 100%;
  background: var(--bg-canvas);
  background-image: radial-gradient(circle, var(--border) 1px, transparent 1px);
  background-size: 20px 20px;
}

.g6-container {
  width: 100%;
  height: 100%;
}

.canvas-toolbar {
  position: absolute;
  top: 0;
  left: 0;
  right: 0;
  z-index: 10;
}

.empty-state {
  position: absolute;
  top: 50%;
  left: 50%;
  transform: translate(-50%, -50%);
  text-align: center;
  color: var(--text-secondary);
}

.empty-icon {

}

  margin-bottom: 20px;
}
</style>
```

### 9.1 工作流 Store

```typescript
// src/stores/workflow.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Workflow, WorkflowNode, WorkflowEdge } from '@/types';
import { workflowApi } from '@/api/workflow';

export const useWorkflowStore = defineStore('workflow', () => {
  // ==================== State ====================
  const workflows = ref<Workflow[]>([]);
  const currentWorkflow = ref<Workflow | null>(null);
  const loading = ref(false);
  const saving = ref(false);

  // 历史记录（撤销/重做）
  const historyStack = ref<{ nodes: WorkflowNode[]; edges: WorkflowEdge[] }[]>([]);
  const historyIndex = ref(-1);

  // ==================== Getters ====================
  const runningWorkflows = computed(() =>
    workflows.value.filter((w) => w.status === 'running')
  );

  const draftWorkflows = computed(() =>
    workflows.value.filter((w) => w.status === 'draft')
  );

  const selectedNode = computed(() => {
    if (!currentWorkflow.value) return null;
    const canvasStore = useCanvasStore();
    return currentWorkflow.value.nodes.find((n) => n.id === canvasStore.selectedNodeId);
  });

  // ==================== Actions ====================
  async function fetchWorkflows() {
    loading.value = true;
    try {
      workflows.value = await workflowApi.getList();
    } finally {
      loading.value = false;
    }
  }

  async function fetchWorkflow(id: string) {
    loading.value = true;
    try {
      currentWorkflow.value = await workflowApi.getById(id);
      // 初始化历史记录
      pushHistory();
    } finally {
      loading.value = false;
    }
  }

  async function saveWorkflow() {
    if (!currentWorkflow.value) return;

    saving.value = true;
    try {
      await workflowApi.update(currentWorkflow.value.id, {
        nodes: currentWorkflow.value.nodes,
        edges: currentWorkflow.value.edges,
      });
      useToast().success('保存成功');
    } finally {
      saving.value = false;
    }
  }

  function addNode(node: WorkflowNode) {
    if (!currentWorkflow.value) return;

    pushHistory();
    currentWorkflow.value.nodes.push(node);
  }

  function updateNode(nodeId: string, updates: Partial<WorkflowNode>) {
    if (!currentWorkflow.value) return;

    pushHistory();
    const index = currentWorkflow.value.nodes.findIndex((n) => n.id === nodeId);
    if (index !== -1) {
      currentWorkflow.value.nodes[index] = {
        ...currentWorkflow.value.nodes[index],
        ...updates,
      };
    }
  }

  function updateNodePosition(nodeId: string, position: { x: number; y: number }) {
    if (!currentWorkflow.value) return;

    const index = currentWorkflow.value.nodes.findIndex((n) => n.id === nodeId);
    if (index !== -1) {
      currentWorkflow.value.nodes[index].position = position;
    }
  }

  function deleteNode(nodeId: string) {
    if (!currentWorkflow.value) return;

    pushHistory();
    // 删除节点
    currentWorkflow.value.nodes = currentWorkflow.value.nodes.filter((n) => n.id !== nodeId);
    // 删除关联边
    currentWorkflow.value.edges = currentWorkflow.value.edges.filter(
      (e) => e.source !== nodeId && e.target !== nodeId
    );
  }

  function addEdge(edge: WorkflowEdge) {
    if (!currentWorkflow.value) return;

    pushHistory();
    currentWorkflow.value.edges.push(edge);
  }

  function deleteEdge(edgeId: string) {
    if (!currentWorkflow.value) return;

    pushHistory();
    currentWorkflow.value.edges = currentWorkflow.value.edges.filter((e) => e.id !== edgeId);
  }

  // ==================== History ====================
  function pushHistory() {
    if (!currentWorkflow.value) return;

    // 移除当前位置之后的历史
    historyStack.value = historyStack.value.slice(0, historyIndex.value + 1);

    // 添加新历史
    historyStack.value.push({
      nodes: JSON.parse(JSON.stringify(currentWorkflow.value.nodes)),
      edges: JSON.parse(JSON.stringify(currentWorkflow.value.edges)),
    });

    historyIndex.value = historyStack.value.length - 1;

    // 限制历史记录数量
    if (historyStack.value.length > 50) {
      historyStack.value.shift();
      historyIndex.value--;
    }
  }

  function undo() {
    if (historyIndex.value > 0) {
      historyIndex.value--;
      restoreHistory();
    }
  }

  function redo() {
    if (historyIndex.value < historyStack.value.length - 1) {
      historyIndex.value++;
      restoreHistory();
    }
  }

  function restoreHistory() {
    if (!currentWorkflow.value) return;

    const history = historyStack.value[historyIndex.value];
    currentWorkflow.value.nodes = JSON.parse(JSON.stringify(history.nodes));
    currentWorkflow.value.edges = JSON.parse(JSON.stringify(history.edges));
  }

  const canUndo = computed(() => historyIndex.value > 0);
  const canRedo = computed(() => historyIndex.value < historyStack.value.length - 1);

  return {
    // State
    workflows,
    currentWorkflow,
    loading,
    saving,

    // Getters
    runningWorkflows,
    draftWorkflows,
    selectedNode,
    canUndo,
    canRedo,

    // Actions
    fetchWorkflows,
    fetchWorkflow,
    saveWorkflow,
    addNode,
    updateNode,
    updateNodePosition,
    deleteNode,
    addEdge,
    deleteEdge,
    undo,
    redo,
  };
});
```

```typescript
  },

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    // 添加 Token
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }

    // 添加 Trace ID
    config.headers['X-Trace-ID'] = generateTraceId();

    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
request.interceptors.response.use(
  (response: AxiosResponse<ApiResponse>) => {
    const { code, message, data } = response.data;

    if (code !== 200) {
      ElMessage.error(message || '请求失败');
      return Promise.reject(new Error(message));
    }

    return data;
  },
  (error) => {
    if (error.response) {
      const { status, data } = error.response;

      switch (status) {
        case 401:
          ElMessage.error('登录已过期，请重新登录');
          localStorage.removeItem('token');
          window.location.href = '/login';
          break;
        case 403:
          ElMessage.error('没有权限');
          break;
        case 404:
          ElMessage.error('资源不存在');
          break;
        case 500:
          ElMessage.error('服务器错误');
          break;
        default:
          ElMessage.error(data?.message || '请求失败');
      }
    } else {
      ElMessage.error('网络错误，请检查网络连接');
    }

    return Promise.reject(error);
  }
);

function generateTraceId(): string {
  return `trace-${Date.now()}-${Math.random().toString(36).slice(2)}`;
}

export default request;
```

```typescript
// src/utils/format.ts
export function format(date: Date, template: string): string {
  const year = date.getFullYear();
  const month = date.getMonth() + 1;
  const day = date.getDate();
  const hour = date.getHours();
  const minute = date.getMinutes();
  const second = date.getSeconds();

  return template
    .replace('YYYY', String(year).padStart(4, '0'))
    .replace('MM', String(month).padStart(2, '0'))
    .replace('DD', String(day).padStart(2, '0'))
    .replace('HH', String(hour).padStart(2, '0'))
    .replace('mm', String(minute).padStart(2, '0'))
    .replace('ss', String(second).padStart(2, '0'));
}

export function formatDuration(ms: number): string {
  if (ms < 1000) return `${ms}ms`;
  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
  if (ms < 3600000) return `${Math.floor(ms / 60000)}m ${Math.floor((ms % 60000) / 1000)}s`;
  return `${Math.floor(ms / 3600000)}h ${Math.floor((ms % 3600000) / 60000)}m`;
}

export function formatCurrency(amount: number): string {
  return `¥${amount.toFixed(2)}`;
}

export function formatNumber(num: number): string {
  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
  return String(num);
}

export function formatRelativeTime(date: Date): string {
  const now = new Date();
  const diff = now.getTime() - date.getTime();

  if (diff < 60000) return '刚刚';
  if (diff < 3600000) return `${Math.floor(diff / 60000)} 分钟前`;
  if (diff < 86400000) return `${Math.floor(diff / 3600000)} 小时前`;
  if (diff < 604800000) return `${Math.floor(diff / 86400000)} 天前`;

  return format(date, 'MM-DD HH:mm');
}
```

### 13.4 不引入 Coze 的部分

- **`@flowgram-adapter`**：Coze 自研的 React 流程图库，与 Vue 不兼容，且 G6 已能 100% 满足需求
- **`fabric`**：Coze 用 fabric.js 做节点布局渲染，G6 5.x 已内置 HTML 节点能力，无需 fabric
- **`Semi Design`**：字节内部 UI 库，社区资源少，迁移到 Element Plus 没必要
- **`Rush Monorepo`**：微软 Rush 工具配置复杂，喵流 50+ 组件规模不需要 monorepo
- **`Storybook`**：组件数量适中，直接在主项目展示即可
