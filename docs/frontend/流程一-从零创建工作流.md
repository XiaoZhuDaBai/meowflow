# 流程一：从零创建工作流

> 详细的前端实现文档，描述从零开始创建工作流的完整流程
> 版本：v1.0 | 更新日期：2026-07-17

---

## 一、流程概述

### 1.1 用户操作路径

```
点击「新建工作流」→ 进入编辑器 → 添加节点 → 配置节点 → 连接节点 → 测试运行 → 保存 → 发布
```

### 1.2 涉及的核心组件

| 组件 | 文件路径 | 职责 |
|------|----------|------|
| `WorkflowList.vue` | `src/views/workflow/List.vue` | 工作流列表页，提供新建入口 |
| `Editor.vue` | `src/views/workflow/Editor.vue` | 可视化编辑器主页面 |
| `WorkflowCanvas.vue` | `src/components/workflow/Canvas/WorkflowCanvas.vue` | 画布核心组件 |
| `NodePalette.vue` | `src/components/workflow/Canvas/NodePalette.vue` | 节点选择面板 |
| `ConfigPanel.vue` | `src/components/workflow/Canvas/ConfigPanel.vue` | 节点配置面板 |
| `CanvasToolbar.vue` | `src/components/workflow/Canvas/CanvasToolbar.vue` | 画布工具栏 |

### 1.3 涉及的数据模型

| 模型 | 文件路径 | 说明 |
|------|----------|------|
| `Workflow` | `src/types/workflow.ts` | 工作流实体 |
| `WorkflowNode` | `src/types/node.ts` | 工作流节点 |
| `WorkflowEdge` | `src/types/node.ts` | 工作流边（连接线） |

// src/views/workflow/Editor.vue
async function handleAddNode(nodeMeta: NodeMeta) {
  // 计算节点位置（画布中心）
  const position = calculateNodePosition();

  // 创建新节点
  const newNode: WorkflowNode = {
    id: `node-${Date.now()}`,
    type: nodeMeta.type,
    category: nodeMeta.category,
    title: nodeMeta.name,
    position,
    config: getDefaultConfig(nodeMeta.type),
    input: { variables: [] },
    output: getDefaultOutput(nodeMeta.type),
  };

  // 添加到 Store
  workflowStore.addNode(newNode);

  // 关闭选择面板
  showPalette.value = false;

  // 自动选中新节点
  selectedNode.value = newNode;
}

function calculateNodePosition(): { x: number; y: number } {
  // 获取画布中心点
  const canvasCenter = canvasRef.value?.getCenter() || { x: 400, y: 300 };
  return {
    x: canvasCenter.x + Math.random() * 100 - 50,
    y: canvasCenter.y + Math.random() * 100 - 50,
  };
}

export const useWorkflowStore = defineStore('workflow', () => {
  const saving = ref(false);

  async function saveWorkflow() {
    if (!currentWorkflow.value) return;

    saving.value = true;
    try {
      await workflowApi.update(currentWorkflow.value.id, {
        name: currentWorkflow.value.name,
        description: currentWorkflow.value.description,

    } catch (error) {
      useToast().error('保存失败');
      throw error;
    } finally {
      saving.value = false;
    }
  }

  return { saving, saveWorkflow, /* ... */ };
});

#### 2.7.2 自动保存

// src/composables/useAutoSave.ts
import { watch, debounce } from 'vue';
import { useWorkflowStore } from '@/stores/workflow';

export function useAutoSave() {
  const store = useWorkflowStore();
  const hasUnsavedChanges = ref(false);

  // 监听节点和边的变化
  watch(
    () => [store.currentWorkflow?.nodes, store.currentWorkflow?.edges],
    debounce(() => {
      hasUnsavedChanges.value = true;
    }, 1000),
    { deep: true }
  );

  // 自动保存
  watch(hasUnsavedChanges, async (changed) => {
    if (changed) {
      await store.saveWorkflow();
      hasUnsavedChanges.value = false;
    }
  });

  return { hasUnsavedChanges };
}
```

    // 2. 调用发布 API
    await workflowApi.publish(workflowId.value);

    // 3. 更新工作流状态
    workflow.value.status = 'running';
    workflow.value.publishedAt = new Date().toISOString();

    // 4. 显示成功提示
    ElMessage.success('发布成功，工作流已开始运行');

    // 5. 如果是 Webhook 触发器，显示调用 URL
    if (hasWebhookTrigger.value) {
      showWebhookUrlDialog();
    }
  } catch (error: any) {
    ElMessage.error(`发布失败: ${error.message}`);
  }
}
```

## 三、Store 状态管理

### 3.1 工作流 Store

```typescript
// src/stores/workflow.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import { workflowApi } from '@/api/workflow';
import type { Workflow, WorkflowNode, WorkflowEdge } from '@/types';

export const useWorkflowStore = defineStore('workflow', () => {
  // State
  const workflows = ref<Workflow[]>([]);
  const currentWorkflow = ref<Workflow | null>(null);
  const selectedNodeId = ref<string | null>(null);
  const loading = ref(false);
  const saving = ref(false);

  // Getters
  const selectedNode = computed(() =>
    currentWorkflow.value?.nodes.find(n => n.id === selectedNodeId.value) || null
  );

  const hasUnsavedChanges = ref(false);

  // Actions
  async function fetchWorkflows() { /* ... */ }
  async function fetchWorkflow(id: string) { /* ... */ }
  async function saveWorkflow() { /* ... */ }
  async function publishWorkflow() { /* ... */ }

  function addNode(node: WorkflowNode) {
    if (!currentWorkflow.value) return;
    currentWorkflow.value.nodes.push(node);
    hasUnsavedChanges.value = true;
  }

  function updateNode(nodeId: string, updates: Partial<WorkflowNode>) {
    if (!currentWorkflow.value) return;
    const index = currentWorkflow.value.nodes.findIndex(n => n.id === nodeId);
    if (index !== -1) {
      currentWorkflow.value.nodes[index] = {
        ...currentWorkflow.value.nodes[index],
        ...updates,
      };
      hasUnsavedChanges.value = true;
    }
  }

  function deleteNode(nodeId: string) {
    if (!currentWorkflow.value) return;
    currentWorkflow.value.nodes = currentWorkflow.value.nodes.filter(n => n.id !== nodeId);
    currentWorkflow.value.edges = currentWorkflow.value.edges.filter(
      e => e.source !== nodeId && e.target !== nodeId
    );
    hasUnsavedChanges.value = true;
  }

  function addEdge(edge: WorkflowEdge) {
    if (!currentWorkflow.value) return;
    currentWorkflow.value.edges.push(edge);
    hasUnsavedChanges.value = true;
  }

  function selectNode(nodeId: string | null) {
    selectedNodeId.value = nodeId;
  }

  return {
    workflows,
    currentWorkflow,
    selectedNodeId,
    selectedNode,
    loading,
    saving,
    hasUnsavedChanges,
    fetchWorkflows,
    fetchWorkflow,
    saveWorkflow,
    publishWorkflow,
    addNode,
    updateNode,
    deleteNode,
    addEdge,
    selectNode,
  };
});
```
