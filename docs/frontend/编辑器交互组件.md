```vue
interface VariableItem {
  name: string;

  description?: string;

  nodeId: string;
  nodeTitle: string;

  if (!keyword.value) return props.variables;
```

### 3.4 上游变量推算（composable）

```typescript
// src/composables/useUpstreamVariables.ts
import { computed } from 'vue';
import type { WorkflowNode, WorkflowEdge } from '@/types';

export function useUpstreamVariables(
  currentNodeId: () => string,
  nodes: () => WorkflowNode[],
  edges: () => WorkflowEdge[],
  outputs: Map<string, any>,
) {
  return computed(() => {
    const id = currentNodeId();
    // 1. 从 DAG 反向遍历所有祖先节点
    const ancestors = new Set<string>();
    const stack = [id];
    while (stack.length) {
      const cur = stack.pop()!;
      edges.value
        .filter(e => e.target === cur)
        .forEach(e => {
          if (!ancestors.has(e.source)) {
            ancestors.add(e.source);
            stack.push(e.source);
          }
        });
    }

    // 2. 收集每个祖先节点的输出变量
    return [...ancestors].map(nodeId => {
      const node = nodes.value.find(n => n.id === nodeId)!;
      const out = outputs.get(nodeId) || node.output || { variables: [] };
      return {
        nodeId,
        nodeTitle: node.title,
        category: node.category,
        variables: out.variables,
      };
    });
  });
}
```

### 3.5 Coze 设计参考（变量系统）

[Coze Studio 的 variable 包](https://github.com/coze-dev/coze-studio/tree/main/frontend/packages/workflow/variable/src) 把变量抽象为 4 层：

```
┌─────────────────────────────────────────────────────────────┐
│ Coze Variable 四层架构                                       │
├─────────────────────────────────────────────────────────────┤
│ Layer 1 - VariableType:   变量类型枚举（String/Number/...） │
│ Layer 2 - VariableField:  字段定义（type/required/default）│
│ Layer 3 - VariableData:   变量数据本体（运行时值）          │
│ Layer 4 - VariableDSL:    JSON 序列化结构                   │
└─────────────────────────────────────────────────────────────┘
```

喵流对应实现：

| Coze | 喵流 | 说明 |
|------|-------|------|
| `VariableType` 枚举 | `src/types/variable.ts` | `string \| number \| boolean \| object \| array` |
| `VariableField` | `src/components/setters/VariableInput.vue` | 字段编辑器（带 picker） |
| `VariableData` | `composables/useUpstreamVariables.ts` | 运行时值推算 |
| `VariableDSL` | 工作流 JSON 内的 `output.variables` | 序列化结构 |

**Coze 的 `FlowNodeVariableData` 关键设计**：

```typescript
// 参考 Coze 源码
interface FlowNodeVariableData {
  // 变量元数据
  name: string;
  type: VariableType;
  required?: boolean;
  defaultValue?: unknown;
  description?: string;

  // 嵌套对象的子字段（用于 object 类型）
  children?: FlowNodeVariableData[];

  // 数组元素的子字段（用于 array 类型）
  item?: FlowNodeVariableData;

  // 来源标识（系统变量 vs 节点输出 vs 用户输入）
  source?: 'system' | 'node' | 'user';
}
```

喵流 `PortVariable` 应该对齐这个结构（已在 `节点组件实现.md` 中定义，需要增强 `children` 和 `item` 字段）。

```typescript
function buildTree(groups: VariableGroup[]) {
  return groups.map(g => ({
    label: g.nodeTitle,
    children: g.variables.map(v => variableToTreeNode(v, `${g.nodeId}.${v.name}`))
  }));
}

function variableToTreeNode(v: PortVariable, path: string): any {
  const node = {
    label: v.name,
    path,
    type: v.type,
    description: v.description,
  };
  if (v.children) {
    node.children = v.children.map(c => variableToTreeNode(c, `${path}.${c.name}`));
  }
  if (v.item) {
    node.children = [variableToTreeNode(v.item, `${path}[]`)];
  }
  return node;
}
```

## 四、ConditionBuilder - 条件构建器

### 4.1 需求场景

条件分支节点的可视化条件编辑，支持 AND/OR 嵌套：

```
┌─────────────────────────────────────────────────────────────┐
│ 条件判断                                                    │
├─────────────────────────────────────────────────────────────┤
│ ( [priority  ▼]  [等于 ▼]  [高     ▼] )  AND ▼   [+ 添加]   │
│ ( [type     ▼]  [包含 ▼]  [发货   ▼] )            [×]       │
│                                                             │
│ 高级模式 [关闭]   │ 表达式：                                │
│                   │ (priority == "高") AND (type ~ "发货") │
└─────────────────────────────────────────────────────────────┘
```

### 4.2 实现

```vue
<!-- src/components/workflow/ConditionBuilder.vue -->
<template>
  <div class="condition-builder">
    <div v-for="(group, gi) in modelValue" :key="gi" class="cond-group">
      <div class="group-op">
        <el-select v-model="group.op" size="small" style="width:80px" @change="emitChange">
          <el-option label="AND" value="AND" />
          <el-option label="OR" value="OR" />
        </el-select>
        <span class="op-hint">{{ gi === 0 ? '满足' : '与/或' }}</span>
      </div>

      <div
        v-for="(item, ii) in group.conditions"
        :key="ii"
        class="cond-row"
      >
        <el-input
          v-model="item.left"
          placeholder="变量或值，如 priority"
          size="small"
          class="cond-left"
        >
          <template #append>
            <el-button
              text
              @click="openVarPicker(item, 'left')"
              :icon="Plus"
            />
          </template>
        </el-input>

        <el-select v-model="item.operator" size="small" class="cond-op">
          <el-option
            v-for="op in operators"
            :key="op.value"
            :label="op.label"
            :value="op.value"
          />
        </el-select>

        <el-input
          v-model="item.right"
          placeholder="比较值"
          size="small"
          class="cond-right"
        />

        <el-button text :icon="Delete" @click="removeCondition(gi, ii)" />
      </div>

      <el-button text size="small" @click="addCondition(gi)">
        <i class="fas fa-plus" /> 添加条件
      </el-button>
      <el-button text size="small" type="danger" @click="removeGroup(gi)">
        <i class="fas fa-trash" /> 删除分组
      </el-button>
    </div>

    <el-button size="small" @click="addGroup">
      <i class="fas fa-plus" /> 添加条件组
    </el-button>

    <el-divider />

    <div class="expr-preview">
      <div class="expr-label">生成表达式（高级模式可手动编辑）：</div>
      <el-input
        :model-value="generatedExpr"
        type="textarea"
        :rows="3"
        readonly
        size="small"
        font-family="monospace"
      />
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue';
import { Delete, Plus } from '@element-plus/icons-vue';

interface Condition {
  left: string;
  operator: string;
  right: string;
}

interface ConditionGroup {
  op: 'AND' | 'OR';
  conditions: Condition[];
}

interface Props {
  modelValue: ConditionGroup[];
}

const props = defineProps<Props>();
const emit = defineEmits<{
  (e: 'update:modelValue', v: ConditionGroup[]): void;
}>();

const operators = [
  { label: '等于', value: '==' },
  { label: '不等于', value: '!=' },
  { label: '大于', value: '>' },
  { label: '大于等于', value: '>=' },
  { label: '小于', value: '<' },
  { label: '小于等于', value: '<=' },
  { label: '包含', value: '~' },
  { label: '不包含', value: '!~' },
  { label: '为空', value: 'is_empty' },
  { label: '不为空', value: 'not_empty' },
];

function emitChange() {
  emit('update:modelValue', [...props.modelValue]);
}

function addGroup() {
  props.modelValue.push({ op: 'AND', conditions: [{ left: '', operator: '==', right: '' }] });
  emitChange();
}

function addCondition(gi: number) {
  props.modelValue[gi].conditions.push({ left: '', operator: '==', right: '' });
  emitChange();
}

function removeCondition(gi: number, ii: number) {
  props.modelValue[gi].conditions.splice(ii, 1);
  if (!props.modelValue[gi].conditions.length) removeGroup(gi);
  else emitChange();
}

function removeGroup(gi: number) {
  props.modelValue.splice(gi, 1);
  emitChange();
}

const generatedExpr = computed(() => {
  return props.modelValue.map((g, gi) => {
    const parts = g.conditions.map(c => {
      if (c.operator === 'is_empty') return `(${c.left} == null)`;
      if (c.operator === 'not_empty') return `(${c.left} != null)`;
      return `(${c.left} ${c.operator} ${JSON.stringify(c.right)})`;
    }).join(` ${g.op} `);
    return gi === 0 ? parts : `${g.op} (${parts})`;
  }).join(' AND ');
});
</script>
```

## 十一、HistoryPanel - 撤销/重做历史

### 11.1 Store 增强

```typescript
// src/stores/workflow.ts 增强
import { defineStore } from 'pinia';
import { ref } from 'vue';

export const useWorkflowStore = defineStore('workflow', () => {
  const current = ref<Workflow | null>(null);
  const history = ref<HistoryItem[]>([]);
  const historyIndex = ref(-1);

  interface HistoryItem {
    id: string;
    timestamp: number;
    action: string;          // 'add-node' | 'delete-node' | 'update-config' | ...
    description: string;     // '添加 LLM 节点'
    snapshot: Workflow;      // 深拷贝的快照
  }

  const MAX_HISTORY = 50;

  function snapshot(action: string, description: string) {
    if (!current.value) return;
    // 截断 redo 分支
    history.value = history.value.slice(0, historyIndex.value + 1);
    history.value.push({
      id: `h-${Date.now()}`,
      timestamp: Date.now(),
      action,
      description,
      snapshot: JSON.parse(JSON.stringify(current.value)),
    });
    if (history.value.length > MAX_HISTORY) history.value.shift();
    historyIndex.value = history.value.length - 1;
  }

  function undo() {
    if (historyIndex.value <= 0) return;
    historyIndex.value--;
    current.value = JSON.parse(JSON.stringify(history.value[historyIndex.value].snapshot));
  }

  function redo() {
    if (historyIndex.value >= history.value.length - 1) return;
    historyIndex.value++;
    current.value = JSON.parse(JSON.stringify(history.value[historyIndex.value].snapshot));
  }

  return { current, history, historyIndex, snapshot, undo, redo };
});
```

### 11.2 面板组件

```vue
<!-- src/components/workflow/HistoryPanel.vue -->

          :key="item.id"

          :hollow="idx !== store.historyIndex"
        >

}
```

## 十二、AutoLayout - 自动布局

### 12.1 实现

```typescript
// src/components/workflow/canvas/autoLayout.ts
import { DagreLayout, CompactBoxLayout } from '@antv/layout';
import type { Graph } from '@antv/g6';

export async function runAutoLayout(graph: Graph, type: 'dagre' | 'compact-box' = 'dagre') {
  const data = graph.getData();
  const nodes = data.nodes || [];
  const edges = data.edges || [];

  const Layout = type === 'dagre' ? DagreLayout : CompactBoxLayout;
  const layout = new Layout({
    type: type === 'dagre' ? 'dagre' : 'compact-box',
    rankdir: 'TB',           // 从上到下
    nodesep: 60,
    ranksep: 80,
    begin: [40, 40],
  });

  const result = await layout.layout({ nodes, edges });

  // 把位置写回 G6
  result.nodes.forEach((n: any) => {
    const node = graph.findById(n.id);
    if (node) graph.updateNodeData(node.id, { style: { x: n.x, y: n.y } });
  });
  graph.render();
}
```

```typescript
// src/composables/useShortcuts.ts
import { onMounted, onBeforeUnmount } from 'vue';
import { useEventListener } from '@vueuse/core';

const shortcuts = [
  { keys: ['Ctrl', 'Z'], desc: '撤销', action: 'undo' },
  { keys: ['Ctrl', 'Shift', 'Z'], desc: '重做', action: 'redo' },
  { keys: ['Delete'], desc: '删除选中', action: 'delete' },
  { keys: ['Ctrl', 'D'], desc: '复制节点', action: 'duplicate' },
  { keys: ['Ctrl', 'C'], desc: '复制', action: 'copy' },
  { keys: ['Ctrl', 'V'], desc: '粘贴', action: 'paste' },
  { keys: ['Space'], desc: '按住拖动画布', action: 'pan' },
  { keys: ['Ctrl', 'S'], desc: '保存', action: 'save' },
  { keys: ['Ctrl', 'Enter'], desc: '试运行', action: 'run' },
  { keys: ['?'], desc: '显示快捷键', action: 'shortcuts' },
];

export function useShortcuts(handlers: Record<string, () => void>) {
  function onKey(e: KeyboardEvent) {
    const pressed = [
      e.ctrlKey && 'Ctrl',
      e.shiftKey && 'Shift',
      e.altKey && 'Alt',
      e.metaKey && 'Meta',
      e.key.toUpperCase(),
    ].filter(Boolean) as string[];

    const match = shortcuts.find(s =>
      s.keys.length === pressed.length &&
      s.keys.every(k => pressed.includes(k))
    );
    if (match && handlers[match.action]) {
      e.preventDefault();
      handlers[match.action]();
    }
  }
  onMounted(() => window.addEventListener('keydown', onKey));
  onBeforeUnmount(() => window.removeEventListener('keydown', onKey));
  return { shortcuts };
}
```

## 十八、Coze Studio 设计哲学小结（参考）

经过对 [coze-studio](https://github.com/coze-dev/coze-studio) 的分析，提取对喵流有指导意义的设计模式：

### 18.1 包拆分原则

| Coze 包 | 职责 | 喵流映射 |
|---------|------|-----------|
| `@coze-workflow/base` | 类型 + 节点元数据 | `src/types/node.ts` + 后端 schema 接口 |
| `@coze-workflow/components` | 通用组件 | `src/components/common/` |
| `@coze-workflow/fabric-canvas` | 画布引擎 + Storybook | G6 + `src/components/workflow/canvas/` |
| `@coze-workflow/setters` | 表单 setter 抽象 | **新增** `src/components/setters/` |
| `@coze-workflow/variable` | 变量系统 | `src/composables/useUpstreamVariables` + 升级 `VariablePicker` |
| `@coze-workflow/test-run` | 试运行 + 测试集 | **新增** `src/components/testrun/` |
| `@coze-workflow/nodes` | 节点定义库 | `src/components/workflow/nodes/` |

### 18.2 复用 Coze 三个关键设计

1. **Setters 抽象** — 节点配置不再为每个节点写表单（详见 `调试与运行组件.md` 第十四章）
2. **Testset 测试集** — 批量用例、批量对比（详见 `调试与运行组件.md` 第十五章）
3. **Variable 四层模型** — Type/Field/Data/DSL 分离（详见 `编辑器交互组件.md` 3.5 节）

### 18.3 暂不引入 Coze 的内容

| Coze 特性 | 暂不引入原因 |
|----------|------------|
| `@flowgram-adapter/free-layout-editor` | 已被 G6 5.x 替代，迁移成本高 |
| `fabric` 自定义画布 | G6 足够应付需求 |
| `Semi Design` UI 库 | 已有 Element Plus，无需替换 |
| `@formily/react` | 改用 `@formily/vue`（Vue 生态） |
| `Rush monorepo` | 喵流项目结构简单，pnpm workspace 足够 |
| `Storybook` | 内部项目无需独立文档站 |
