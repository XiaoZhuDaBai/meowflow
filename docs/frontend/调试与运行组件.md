## 一、组件清单

| # | 组件 | 用途 | 依赖库 | 优先级 |
|---|------|------|--------|--------|
| 1 | TestRunDialog | 测试运行对话框（填入模拟输入） | Element Plus | P0 |
| 2 | LogStream | 实时日志流（WebSocket） | Element Plus + WebSocket | P0 |
| 3 | NodeRunDetail | 单节点执行详情面板 | Element Plus | P0 |

| 7 | NodeStatusBadge | 节点状态徽章 | Element Plus | P1 |
| 8 | ExecutionHistory | 执行历史列表 | Element Plus | P2 |
| 9 | StepDebugger | 单步调试器 | Element Plus | P2 |

```vue
import { ref, computed, watch } from 'vue';
import { ElMessage } from 'element-plus';
import { useWorkflowStore } from '@/stores/workflow';
import { testRun } from '@/api/workflow';
import LogStream from './LogStream.vue';

const visible = defineModel<boolean>({ default: false });
const workflowStore = useWorkflowStore();

const startNodeId = ref('');
const inputJson = ref('{}');
const autoScroll = ref(true);
const running = ref(false);
const currentExecutionId = ref('');

const triggerNodes = computed(() =>
  workflowStore.current?.nodes.filter(n => n.category === 'trigger') || []
);

const isValid = computed(() => {
  try { JSON.parse(inputJson.value); return true; } catch { return false; }
});

watch(visible, (v) => {
  if (v) {
    const first = triggerNodes.value[0];
    if (first) {
      startNodeId.value = first.id;
      generateSample();
    }
  }

function generateSample() {
  const node = workflowStore.current?.nodes.find(n => n.id === startNodeId.value);
  // 根据节点类型生成示例
  const samples: Record<string, any> = {
    webhook: { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: { query: '示例问题' } },
    schedule: { timestamp: Date.now() },
    form: { fields: {} },
    message: { content: '示例消息', from: 'user-001' },
  };

async function pasteFromClipboard() {
  try {
    const text = await navigator.clipboard.readText();
    inputJson.value = text;
  } catch {
    ElMessage.warning('剪贴板读取失败，请检查浏览器权限');
  }
}

function formatJson() {
  try {
    inputJson.value = JSON.stringify(JSON.parse(inputJson.value), null, 2);
  } catch {}
}

async function onRun() {
  if (!isValid.value) return ElMessage.error('JSON 格式错误');
  if (!startNodeId.value) return ElMessage.warning('请选择起始节点');

  running.value = true;
  try {
    const res = await testRun({
      workflowId: workflowStore.current!.id,
      startNodeId: startNodeId.value,
      input: JSON.parse(inputJson.value),
    });
    currentExecutionId.value = res.executionId;
  } catch (e: any) {
    ElMessage.error(e.message || '运行失败');
  } finally {
    running.value = false;
  }
}
</script>

  }
```

## 四、LogStream - 实时日志流

### 4.1 WebSocket 连接管理

```typescript
// src/composables/useExecutionWebSocket.ts
import { ref, onUnmounted } from 'vue';

export interface LogEntry {
  timestamp: number;
  level: 'info' | 'success' | 'warning' | 'error' | 'debug';
  nodeId?: string;
  nodeTitle?: string;
  message: string;
  data?: any;
}

export function useExecutionWebSocket() {
  const logs = ref<LogEntry[]>([]);
  const status = ref<'idle' | 'connecting' | 'running' | 'completed' | 'failed'>('idle');
  const progress = ref(0);
  const currentNodeId = ref<string | null>(null);
  let ws: WebSocket | null = null;

  function connect(executionId: string) {
    const baseURL = import.meta.env.VITE_WS_BASE_URL || 'ws://localhost:8080/ws';
    const url = `${baseURL}/executions/${executionId}/logs?token=${localStorage.getItem('token')}`;
    status.value = 'connecting';
    ws = new WebSocket(url);

    ws.onopen = () => { status.value = 'running'; };
    ws.onmessage = (e) => {
      const msg = JSON.parse(e.data);
      switch (msg.type) {
        case 'log':
          logs.value.push(msg.payload);
          break;
        case 'progress':
          progress.value = msg.payload.percent;
          currentNodeId.value = msg.payload.nodeId;
          break;
        case 'node-status':
          // 触发节点状态变化
          break;
        case 'completed':
          status.value = 'completed';
          break;
        case 'failed':
          status.value = 'failed';
          break;
      }
    };
    ws.onerror = () => { status.value = 'failed'; };
    ws.onclose = () => { /* keep logs */ };
  }

  function close() {
    ws?.close();
    ws = null;
  }

  function clear() {
    logs.value = [];
    progress.value = 0;
  }

  onUnmounted(() => close());

  return { logs, status, progress, currentNodeId, connect, close, clear };
}
```

```typescript
  onChange?: (value: V) => void;

  readonly?: boolean;
```

```typescript
    "method": { "type": "enum", "options": ["GET", "POST", "PUT", "DELETE"] },
```

```vue
<!-- src/components/testrun/InputFormEngine.vue -->
<template>
  <FormProvider :form="form">
    <SchemaField
      :schema="{ type: 'object', properties: properties }"
    />
  </FormProvider>
</template>

<script setup lang="ts">
import { createForm, onFormValuesChange } from '@formily/core';
import { FormProvider, SchemaField } from '@formily/vue';
import { computed } from 'vue';
import { getSetter } from '@/components/setters/registerSetter';

interface Props {
  schema: any;
  initialValues: Record<string, any>;
}

const props = defineProps<Props>();
const emit = defineEmits<{ (e: 'change', v: any): void }>();

const form = createForm({
  values: props.initialValues,
  effects: () => {
    onFormValuesChange((values) => emit('change', values));
  },
});

const properties = computed(() => {
  const result: Record<string, any> = {};
  for (const [key, def] of Object.entries(props.schema.properties || {})) {
    result[key] = {
      type: 'string',
      title: (def as any).label || key,
      'x-component': getSetter((def as any).type),
      'x-component-props': def,
    };
  }
  return result;
});
</script>
```
