## 二、类型定义

### 2.1 节点基础类型

```typescript
// src/types/node.ts

/** 节点分类 */
export enum NodeCategory {
  TRIGGER = 'trigger',
  AI = 'ai',
  FLOW = 'flow',
  TOOL = 'tool',
  NOTIFY = 'notify',
}

/** 节点类型 */
export enum NodeType {
  // 触发器
  WEBHOOK = 'webhook',
  SCHEDULE = 'schedule',
  FORM = 'form',
  MESSAGE = 'message',

  // AI
  LLM = 'llm',
  CLASSIFY = 'classify',
  EXTRACT = 'extract',
  SUMMARIZE = 'summarize',
  KNOWLEDGE_SEARCH = 'knowledge-search',

  // 流程控制
  CONDITION = 'condition',
  LOOP = 'loop',
  PARALLEL = 'parallel',
  DELAY = 'delay',

  // 工具
  HTTP = 'http',
  DATABASE = 'database',
  CODE = 'code',
  VARIABLE = 'variable',

  // 通知
  DINGTALK = 'dingtalk',
  WECHAT = 'wechat',
  EMAIL = 'email',
  SMS = 'sms',
}

/** 节点状态 */
export type NodeStatus = 'idle' | 'running' | 'success' | 'error' | 'waiting';

/** 节点配置基类 */
export interface BaseNodeConfig {
  timeout?: number;      // 超时时间（秒）
  retry?: number;       // 重试次数
  description?: string;  // 描述
}

/** 节点端口变量 */
export interface PortVariable {
  name: string;           // 变量名
  type: 'string' | 'number' | 'boolean' | 'object' | 'array';  // 类型
  description?: string;   // 描述
  required?: boolean;     // 是否必填
  defaultValue?: any;     // 默认值
}

/** 节点端口 */
export interface NodePort {
  variables: PortVariable[];
}

/** 工作流节点 */
export interface WorkflowNode {
  id: string;
  type: NodeType;
  category: NodeCategory;
  title: string;
  position: { x: number; y: number };
  config: Record<string, any>;
  input?: NodePort;
  output?: NodePort;
  status?: NodeStatus;
  // 条件分支专用
  branches?: ConditionBranch[];
}

/** 条件分支 */
export interface ConditionBranch {
  id: string;
  label: string;
  condition: string;       // 条件表达式
  description?: string;
}

/** 工作流边 */
export interface WorkflowEdge {
  id: string;
  source: string;           // 源节点ID
  target: string;           // 目标节点ID
  sourceHandle?: string;     // 源端口（条件分支用）
  targetHandle?: string;     // 目标端口
  label?: string;           // 边标签
}
```

### 4.2 基础节点组件

```vue
<!-- src/components/workflow/nodes/BaseNode.vue -->
<template>
  <div
    class="base-node"
    :class="[`node-${category}`, { selected, running, error: status === 'error' }]"
    @mousedown="handleMouseDown"
    @click="handleClick"
  >
    <div class="node-header">
      <div class="node-icon" :style="{ background: color }">
        <i :class="`fas fa-${icon}`" />
      </div>
      <div class="node-title">{{ title }}</div>
      <div v-if="showStatus" class="node-status-indicator">
        <span v-if="status === 'running'" class="status-dot running" />
        <span v-else-if="status === 'success'" class="status-dot success" />
        <span v-else-if="status === 'error'" class="status-dot error" />
        <span v-else-if="status === 'waiting'" class="status-dot waiting" />
      </div>
    </div>
    <div class="node-body">
      <slot name="body">
        <div class="node-preview">{{ preview }}</div>
      </slot>
    </div>

    <!-- 输入端口 -->
    <div
      v-if="showInputPort"
      class="node-port input-port"
      @mouseup="handlePortMouseUp('input', $event)"
    >
      <div class="port-dot" />
    </div>

    <!-- 输出端口 -->
    <div
      v-if="showOutputPort"
      class="node-port output-port"
      @mousedown="handlePortMouseDown('output', $event)"
    >
      <div class="port-dot" />
    </div>

    <!-- 多输出端口（条件分支等） -->
    <div
      v-if="showMultiOutput"
      v-for="(branch, index) in branches"
      :key="branch.id"
      class="node-port output-port multi"
      :style="{ top: `${40 + index * 24}px` }"
      @mousedown="handlePortMouseDown('output', $event, branch.id)"
    >
      <div class="port-dot" :style="{ background: branch.color || color }" />
      <span class="port-label">{{ branch.label }}</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue';
import { NODE_META_MAP } from '@/config/nodes';
import type { WorkflowNode, ConditionBranch } from '@/types/node';

interface Props {
  node: WorkflowNode;
  selected?: boolean;
  showInputPort?: boolean;
  showOutputPort?: boolean;
  showMultiOutput?: boolean;
  branches?: ConditionBranch[];
}

const props = withDefaults(defineProps<Props>(), {
  selected: false,
  showInputPort: true,
  showOutputPort: true,
  showMultiOutput: false,
  branches: () => [],
});

const emit = defineEmits<{
  (e: 'click', node: WorkflowNode): void;
  (e: 'dragstart', event: MouseEvent, node: WorkflowNode): void;
  (e: 'portdragstart', type: 'input' | 'output', branchId?: string, event?: MouseEvent): void;
}>();

const meta = computed(() => NODE_META_MAP[props.node.type]);

const category = computed(() => props.node.category);
const icon = computed(() => meta.value?.icon || 'circle');
const color = computed(() => meta.value?.color || '#666');
const title = computed(() => props.node.title || meta.value?.name || '');
const status = computed(() => props.node.status);

const showStatus = computed(() => status.value && status.value !== 'idle');

const preview = computed(() => {
  const config = props.node.config;
  if (!config) return '';

  // 根据节点类型显示预览信息
  switch (props.node.type) {
    case 'llm':
    case 'classify':
    case 'extract':
    case 'summarize':
      return config.model || 'GPT-4o';
    case 'http':
      return `${config.method || 'GET'} ${config.url || ''}`;
    case 'delay':
      return `${config.duration || 0}${config.unit || 's'}`;
    case 'webhook':
      return config.path || '/webhook';
    case 'schedule':
      return config.cron || '* * * * *';
    default:
      return '';
```

### 6.1 节点执行状态管理

```typescript
// src/composables/useNodeExecution.ts
import { ref, computed } from 'vue';
import type { WorkflowNode, NodeStatus } from '@/types/node';

interface ExecutionLog {
  nodeId: string;
  status: NodeStatus;
  startTime: number;
  endTime?: number;
  duration?: number;
  input?: any;
  output?: any;
  error?: string;
}

export function useNodeExecution() {
  const executionLogs = ref<Map<string, ExecutionLog>>(new Map());
  const currentExecutingNodeId = ref<string | null>(null);

  /** 开始执行节点 */
  function startNode(nodeId: string, input?: any) {
    currentExecutingNodeId.value = nodeId;
    executionLogs.value.set(nodeId, {
      nodeId,
      status: 'running',
      startTime: Date.now(),
      input,
    });
  }

  /** 节点执行成功 */
  function finishNode(nodeId: string, output?: any) {
    const log = executionLogs.value.get(nodeId);
    if (log) {
      const endTime = Date.now();
      executionLogs.value.set(nodeId, {
        ...log,
        status: 'success',
        endTime,
        duration: endTime - log.startTime,
        output,
      });
    }
    if (currentExecutingNodeId.value === nodeId) {
      currentExecutingNodeId.value = null;
    }
  }

  /** 节点执行失败 */
  function failNode(nodeId: string, error: string) {
    const log = executionLogs.value.get(nodeId);
    if (log) {
      const endTime = Date.now();
      executionLogs.value.set(nodeId, {
        ...log,
        status: 'error',
        endTime,
        duration: endTime - log.startTime,
        error,
      });
    }
    if (currentExecutingNodeId.value === nodeId) {
      currentExecutingNodeId.value = null;
    }
  }

  /** 节点等待人工处理 */
  function waitNode(nodeId: string) {
    const log = executionLogs.value.get(nodeId);
    if (log) {
      executionLogs.value.set(nodeId, {
        ...log,
        status: 'waiting',
      });
    }
  }

  /** 获取节点状态 */
  function getNodeStatus(nodeId: string): NodeStatus | undefined {
    return executionLogs.value.get(nodeId)?.status;
  }

  /** 获取节点执行日志 */
  function getNodeLog(nodeId: string): ExecutionLog | undefined {
    return executionLogs.value.get(nodeId);
  }

  /** 清空执行日志 */
  function clearLogs() {
    executionLogs.value.clear();
    currentExecutingNodeId.value = null;
  }

  /** 获取所有执行中的节点 */
  const runningNodes = computed(() => {
    return Array.from(executionLogs.value.values())
      .filter(log => log.status === 'running' || log.status === 'waiting')
      .map(log => log.nodeId);
  });

  return {
    executionLogs,
    currentExecutingNodeId,
    runningNodes,
    startNode,
    finishNode,
    failNode,
    waitNode,
    getNodeStatus,
    getNodeLog,
    clearLogs,
  };
}
```

### 7.1 节点操作 Hook

```typescript
// src/composables/useNodeDrag.ts
import { ref, computed } from 'vue';
import { useWorkflowStore } from '@/stores/workflow';
import { useCanvasStore } from '@/stores/canvas';
import type { WorkflowNode, WorkflowEdge } from '@/types/node';
import { getNodeComponent } from '@/components/workflow/nodes/NodeRenderer';

export function useNodeDrag() {
  const workflowStore = useWorkflowStore();
  const canvasStore = useCanvasStore();

  const isDraggingNode = ref(false);
  const draggedNodeId = ref<string | null>(null);
  const dragOffset = ref({ x: 0, y: 0 });

  const isDrawingEdge = ref(false);
  const edgeStartNode = ref<string | null>(null);
  const edgeStartPort = ref<string | null>(null);
  const edgeEndPosition = ref({ x: 0, y: 0 });

  /** 开始拖拽节点 */
  function startNodeDrag(nodeId: string, event: MouseEvent) {
    isDraggingNode.value = true;
    draggedNodeId.value = nodeId;

    const node = workflowStore.currentWorkflow?.nodes.find(n => n.id === nodeId);
    if (node) {
      dragOffset.value = {
        x: event.clientX - node.position.x,
        y: event.clientY - node.position.y,
      };
    }
  }

  /** 拖拽节点移动 */
  function onNodeDrag(event: MouseEvent) {
    if (!isDraggingNode.value || !draggedNodeId.value) return;

    const newPosition = {
      x: event.clientX - dragOffset.value.x,
      y: event.clientY - dragOffset.value.y,
    };

    workflowStore.updateNodePosition(draggedNodeId.value, newPosition);
  }

  /** 结束拖拽节点 */
  function endNodeDrag() {
    isDraggingNode.value = false;
    draggedNodeId.value = null;
    dragOffset.value = { x: 0, y: 0 };
  }

  /** 开始绘制连线 */
  function startEdgeDraw(nodeId: string, portId?: string, event?: MouseEvent) {
    isDrawingEdge.value = true;
    edgeStartNode.value = nodeId;
    edgeStartPort.value = portId || null;

    if (event) {
      edgeEndPosition.value = { x: event.clientX, y: event.clientY };
    }
  }

  /** 更新连线终点 */
  function updateEdgeDraw(event: MouseEvent) {
    if (!isDrawingEdge.value) return;
    edgeEndPosition.value = { x: event.clientX, y: event.clientY };
  }

  /** 完成连线绘制 */
  function finishEdgeDraw(targetNodeId: string, targetPortId?: string) {
    if (!isDrawingEdge.value || !edgeStartNode.value) return;

    // 防止连接到自己
    if (edgeStartNode.value === targetNodeId) {
      cancelEdgeDraw();
      return;
    }

    // 创建新的边
    const newEdge: WorkflowEdge = {
      id: `edge-${Date.now()}`,
      source: edgeStartNode.value,
      target: targetNodeId,
      sourceHandle: edgeStartPort.value || undefined,
      targetHandle: targetPortId || undefined,
    };

    // 检查是否已存在相同的边
    const exists = workflowStore.currentWorkflow?.edges.some(
      e => e.source === newEdge.source && e.target === newEdge.target
    );

    if (!exists) {
      workflowStore.addEdge(newEdge);
    }

    cancelEdgeDraw();
  }

  /** 取消连线绘制 */
  function cancelEdgeDraw() {
    isDrawingEdge.value = false;
    edgeStartNode.value = null;
    edgeStartPort.value = null;
    edgeEndPosition.value = { x: 0, y: 0 };
  }

  /** 删除边 */
  function deleteEdge(edgeId: string) {
    workflowStore.deleteEdge(edgeId);
  }

  /** 删除节点及其关联边 */
  function deleteNode(nodeId: string) {
    workflowStore.deleteNode(nodeId);
  }

  return {
    // 节点拖拽
    isDraggingNode,
    draggedNodeId,
    startNodeDrag,
    onNodeDrag,
    endNodeDrag,

    // 边绘制
    isDrawingEdge,
    edgeStartNode,
    edgeEndPosition,
    startEdgeDraw,
    updateEdgeDraw,
    finishEdgeDraw,
    cancelEdgeDraw,

    // 操作
    deleteEdge,
    deleteNode,
  };
}
```
